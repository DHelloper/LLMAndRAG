-- pgvector 확장 활성화
CREATE EXTENSION IF NOT EXISTS vector;
-- 문서 메타데이터 (8주차에 Spring이 조회)
CREATE TABLE documents (
id BIGSERIAL PRIMARY KEY,
filename TEXT NOT NULL,
title TEXT,
pages INT,
created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
-- 청크 + 임베딩 (6주차 색인, 7주차 검색)
-- 384은 6주차에서 쓰는 intfloat/multilingual-e5-small 의 차원 수입니다.
-- 임베딩 모델을 바꾸면 이 숫자도 바꾸고 전체를 재색인해야 합니다.
CREATE TABLE chunks (
id BIGSERIAL PRIMARY KEY,
document_id BIGINT NOT NULL REFERENCES documents(id) ON DELETE CASCADE,
page INT NOT NULL,
chunk_index INT NOT NULL,
content TEXT NOT NULL,
embedding vector(384)
);
-- 코사인 거리 기준 근사 최근접 인덱스
CREATE INDEX chunks_embedding_idx
ON chunks USING hnsw (embedding vector_cosine_ops);
-- 설치 검증: 두 벡터의 코사인 거리 계산
SELECT '[1,0,0]'::vector <=> '[0,1,0]'::vector AS cosine_distance;
-- 기대 결과: 1 (완전히 직교 = 전혀 안 닮음)
